# usePropsRef

Hook that copy an sync the given value in a ref, to be used in a non-reactive way.

```tsx
const MyComponent: React.FC = (props) => {
    const propsCopy = usePropsRef(props);

    const handleClick = useCallback(() => {
        // last props value can be used without the need
        // to add it to the dependency array
        console.log(propsCopy.current);
    }, []);

    return <button onClick={handleClick}></button>;
};
```
